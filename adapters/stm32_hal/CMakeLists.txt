cmake_minimum_required(VERSION 3.22)
foreach(bus QSPI OSPI)
    option(STM_FLASH_WITH_${bus} "Build the STM32 HAL ${bus} adapter" OFF)
    if(STM_FLASH_WITH_${bus})
        string(TOLOWER ${bus} name)
        add_library(stm_flash_${name} STATIC flash_${name}.c)
        target_include_directories(stm_flash_${name} PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
        target_link_libraries(stm_flash_${name} PUBLIC stm_flash)
        if(TARGET stm32cubemx)
            target_link_libraries(stm_flash_${name} PUBLIC stm32cubemx)
        endif()
    endif()
endforeach()
