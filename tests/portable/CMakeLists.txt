cmake_minimum_required(VERSION 3.22)
project(stm_flash_portable_tests C CXX)
enable_testing()
add_subdirectory(../.. component)
add_executable(flash_portable test_core.c headers.cpp)
target_link_libraries(flash_portable PRIVATE stm_flash)
target_compile_features(flash_portable PRIVATE c_std_11 cxx_std_17)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(stm_flash PRIVATE -Wall -Wextra -Werror)
    target_compile_options(flash_portable PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME flash_portable COMMAND flash_portable)

# SPI adapter tests use a deliberately separate fake HAL; the portable core remains HAL-free.
add_executable(flash_gd25q128 test_gd25q128.c)
target_link_libraries(flash_gd25q128 PRIVATE stm_flash)
set(STM_FLASH_WITH_SPI ON)
add_subdirectory(../../adapters/stm32_hal adapters)
target_compile_definitions(stm_flash_spi PUBLIC STM_FLASH_HAL_HEADER="spi_test_hal.h")
target_include_directories(stm_flash_spi PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/fake_hal)
add_executable(flash_spi_test test_spi.c spi_headers.cpp)
target_link_libraries(flash_spi_test PRIVATE stm_flash_spi)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    foreach(t stm_flash_spi flash_gd25q128 flash_spi_test)
        target_compile_options(${t} PRIVATE -Wall -Wextra -Werror)
    endforeach()
endif()
add_test(NAME flash_gd25q128 COMMAND flash_gd25q128)
add_test(NAME flash_spi_test COMMAND flash_spi_test)

# Compile the read-only integration example using the same adapter header contract.
add_library(flash_spi_example OBJECT ../../example/gd25q128_spi.c)
target_link_libraries(flash_spi_example PRIVATE stm_flash_spi)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(flash_spi_example PRIVATE -Wall -Wextra -Werror)
endif()
